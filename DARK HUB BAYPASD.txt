if v >= 1e6  then return string.format("%.2fM", v/1e6)  end

local function MoveTo(target, speed, easeOut)
local _, hrp = getChar()
if not hrp or not target then return false end
local start = hrp.Position
local dist = (target - start).Magnitude
if dist < 1 then
hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
hrp.AssemblyLinearVelocity = Vector3.zero
hrp.AssemblyAngularVelocity = Vector3.zero
return true
end
speed = math.clamp(tonumber(speed) or 750, 50, 750)
local t0 = os.clock()
local totalDist = dist
while not HUB.dead do
local dt = RunService.Heartbeat:Wait()
local cp = hrp.Position
local to = target - cp
local rem = to.Magnitude
if rem < 1 then break end
local sp2 = speed
if easeOut then
local prog = 1 - math.clamp(rem / totalDist, 0, 1)
sp2 = math.max(speed * (1 - prog * 0.8), 35)
end
local step = math.min(sp2 * dt, rem)
local dir = to.Unit
local np = cp + dir * step
hrp.CFrame = CFrame.lookAt(np, np + dir)
hrp.AssemblyLinearVelocity = Vector3.zero
hrp.AssemblyAngularVelocity = Vector3.zero
SetSpoofWalkSpeed(HUB.flySpeed)
if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
SyncHoverPad()
if os.clock() - t0 > (totalDist / 50 + 5) then break end
end
hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
hrp.AssemblyLinearVelocity = Vector3.zero
hrp.AssemblyAngularVelocity = Vector3.zero
return true
end

local function FlyTo(target, speed, easeOut)
local _, hrp = getChar()
if not hrp or not target then return false end
local start = hrp.Position
local dist = (target - start).Magnitude
if dist < 1 then
hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
hrp.AssemblyLinearVelocity = Vector3.zero
hrp.AssemblyAngularVelocity = Vector3.zero
return true
end
speed = math.clamp(tonumber(speed) or 750, 50, 750)
local mt = math.max(dist / speed, 0.02)
if easeOut then mt = mt * 1.25 end
local t0 = os.clock()
local delta = target - start
local dir = delta.Magnitude > 0.001 and delta.Unit or Vector3.new(1, 0, 0)
while os.clock() - t0 < mt and not HUB.dead do
RunService.Heartbeat:Wait()
local a = math.clamp((os.clock() - t0) / mt, 0, 1)
if easeOut then a = math.sin(a * (math.pi / 2)) end
local cur = start:Lerp(target, a)
hrp.CFrame = CFrame.lookAt(cur, cur + dir)
local cs = speed
if easeOut then cs = math.max(speed * (1 - a * 0.8), 35) end
hrp.AssemblyLinearVelocity = Vector3.new(dir.X * cs, math.clamp(dir.Y * cs, -15, 150), dir.Z * cs)
hrp.AssemblyAngularVelocity = Vector3.zero
SetSpoofWalkSpeed(HUB.flySpeed)
if BypassState.integrityState then SpoofState(BypassState.integrityState, HUB.flySpeed * HEADROOM) end
SyncHoverPad()
end
hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
hrp.AssemblyLinearVelocity = Vector3.zero
hrp.AssemblyAngularVelocity = Vector3.zero
return true
end

MoveTo(Vector3.new(sp0.X, sy, ROAD_Z), spd, false)
MoveTo(Vector3.new(SAFE_X, sy, ROAD_Z), spd, false)
MoveTo(Vector3.new(target.X, sy, ROAD_Z), SAFE_SPD, false)

else
MoveTo(Vector3.new(sp0.X, sy, ROAD_Z), spd, false)
MoveTo(Vector3.new(target.X, sy, ROAD_Z), spd, false)
MoveTo(target + Vector3.new(0, 1.2, 0), spd, approach == true)
end

end

stealEgg = function(record)
if not record then return false end
local uid = record.Uid
if not uid then return false end

if record.State == "Carried" and record.CarrierUserId and record.CarrierUserId ~= LP.UserId then
return trackAndSteal(record.CarrierUserId, uid)
end

local targetPos = record.BoundsCFrame and record.BoundsCFrame.Position
    or record.Position
    or (record.CFrame and record.CFrame.Position)
if not targetPos then return false end

flyTo(targetPos + Vector3.new(0, 2, 0), 15)
if HUB.dead then return false end
local _, hrp = getChar()
if not hrp then return false end
local slotKey = nil
if AreaEggSlotId and AreaEggSlotId.LooksLikeFirstAreaUid
and AreaEggSlotId.LooksLikeFirstAreaUid(uid) then
slotKey = AreaEggSlotId.SlotKey(record.AreaId, record.NestId)
end
if EggWorld and EggWorld.AskFieldEggCarry then
pcall(function() EggWorld.AskFieldEggCarry:InvokeServer({ Uid=uid, FirstAreaSlotKey=slotKey }) end)
end
pcall(function()
if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end
end)
local prompt = nil
for _, d in ipairs(Workspace:GetDescendants()) do
if d:IsA("ProximityPrompt") and d.Enabled then
local nm = d.Name
if nm == "CarryAreaEgg" or nm == "PickupEgg" or nm == "CarryEgg" or nm == "GrabEgg" or nm == "Pickup" then
local p = d.Parent
if p and p:IsA("Attachment") then p = p.Parent end
if p and p:IsA("BasePart") and (p.Position - hrp.Position).Magnitude < 18 then
prompt = d break
end
end
end
end
if prompt then
prompt.HoldDuration = 0
pcall(function() fireproximityprompt(prompt) end)
end

local t0 = tick()
while tick() - t0 < 2 and not HUB.dead do
if isCarryingEgg() then
    if doHitTeleport then doHitTeleport() end
    return true
end
pcall(function()
if EggState.CarryFieldEgg then EggState.CarryFieldEgg(uid, slotKey) end

if prompt then pcall(function() fireproximityprompt(prompt) end) end
task.wait(0.1)
end

if isCarryingEgg() then
    if doHitTeleport then doHitTeleport() end
    return true
end
return false
end

local function plantAllEggs()
if not EggState or not EggState.PlantEgg or not EggToolDisplay then return 0 end
local ids = {}
for _, container in ipairs({LP.Character, LP:FindFirstChild("Backpack")}) do
if container then
for _, t in ipairs(container:GetChildren()) do
if t:IsA("Tool") and EggToolDisplay.IsEggTool and EggToolDisplay.IsEggTool(t) then
local uid = EggToolDisplay.GetToolUid and EggToolDisplay.GetToolUid(t)
if uid then table.insert(ids, uid) end
end
end
end
end
local count = 0
for _, uid in ipairs(ids) do
local ok, res = pcall(function()
return EggState.PlantEgg(uid, CFrame.new(math.random(-6,6), 0, math.random(-6,6)))
end)

end

end

doHitTeleport = nil

local function returnHome()
if not isCarryingEgg() then return false end
setStatus(" Plantando...", true)
task.wait(0.3)
plantAllEggs()
return true
end

local function runCycle()
if HUB.busy or HUB.dead or not HUB.running then return end
HUB.busy = true
if isCarryingEgg() then returnHome() HUB.busy = false return end

if HUB.activeTarget and HUB.activeTarget.uid then
local rec = findRecord(HUB.activeTarget.uid)
if rec then
if stealEgg(rec) then HUB.activeTarget = nil returnHome() end
HUB.busy = false return
else HUB.activeTarget = nil end
end

local eggs = getFieldEggs()
if #eggs == 0 then HUB.busy = false return end

local target = nil
if HUB.selectedUid then
for _, e in ipairs(eggs) do
if e.uid == HUB.selectedUid then target = e break end
end
end
if not target then
table.sort(eggs, function(a, b)
if a.value ~= b.value then return a.value > b.value end
return a.rank > b.rank
end)
target = eggs[1]
end

HUB.activeTarget = { uid = target.uid, name = target.name }
if stealEgg(target.record) then
HUB.activeTarget = nil
returnHome()
end
HUB.busy = false
end

local function refreshEggList()
local eggs = getFieldEggs()

if HUB.sortMode == "GEN" then
table.sort(eggs, function(a, b)
if a.value ~= b.value then return a.value > b.value end
return a.rank > b.rank
end)
else
table.sort(eggs, function(a, b)
if a.rank ~= b.rank then return a.rank > b.rank end
return a.value > b.value
end)
end

local seen = {}
local newIcons = {}

for i, e in ipairs(eggs) do
seen[e.uid] = true
local r = rowMap[e.uid]

if not r then
r = createRow(e, i)
if e.icon then newIcons[#newIcons+1] = e.icon end
else
r.frame.LayoutOrder = i
local genText = string.format("Gen %s    KG %s     %s",
fmtValue(e.value), fmtKg(e.kg), e.area)
if r.lastGenText ~= genText then
r.genLbl.Text = genText
r.lastGenText = genText
end
local rarText = "Rarity " .. e.rarity
local rarColor = e.rcolor or RARITY_COLOR[e.rarity] or THEME.textMuted
if r.lastRarText ~= rarText then
r.rarLbl.Text = rarText
r.lastRarText = rarText
end
if r.lastRarColor ~= rarColor then
r.rarLbl.TextColor3 = rarColor
r.lastRarColor = rarColor
end
if r.nameLbl.Text ~= e.name then r.nameLbl.Text = e.name end
if e.icon and r.photo.Image ~= e.icon then
r.photo.Image = e.icon
newIcons[#newIcons+1] = e.icon
end
local picked = (HUB.selectedUid == e.uid)
if picked ~= r.isPicked then
r.isPicked = picked
r.frame.BackgroundColor3 = picked and THEME.rowPicked or THEME.rowNormal
end
end
end

local toRemove = {}
for uid in pairs(rowMap) do
if not seen[uid] then toRemove[#toRemove+1] = uid end
end
for _, uid in ipairs(toRemove) do
local r = rowMap[uid]
if r and r.frame then r.frame:Destroy() end
rowMap[uid] = nil
end

if #newIcons > 0 then
task.spawn(function()
pcall(function() ContentProvider:PreloadAsync(newIcons) end)
end)
end

if HUB.selectedUid and not seen[HUB.selectedUid] then
HUB.selectedUid = nil
if setStatus then setStatus(" Ovo saiu do campo", false) end
end

emptyLbl.Visible = (#eggs == 0)
end

local function showHitLoading(duration)
    duration = duration or 0.9
    hitToken = hitToken + 1
    local myToken = hitToken
    hitGui.Enabled = true
    hitBarFill.Size = UDim2.new(0, 0, 1, 0)
    hitPct.Text = "0%"
    hitSub.Text = "Pegando o ovo"
    local startTime = tick()
    task.spawn(function()
        while hitGui.Enabled and hitToken == myToken do
            local el = tick() - startTime
            local prog = math.clamp(el / duration, 0, 1)
            hitBarFill.Size = UDim2.new(prog, 0, 1, 0)
            hitPct.Text = tostring(math.floor(prog * 100)) .. "%"
            if prog >= 1 then
                hitSub.Text = " Concludo"
                break
            end
            task.wait(0.016)
        end
    end)
end

    hitToken = hitToken + 1
    hitGui.Enabled = false
end
